package main

import (
	"encoding/base64"
	"encoding/json"
	"fmt"
	"os"
	"runtime"

	"github.com/grafana/ai-sdk/gateway/providerwire"
	"github.com/grafana/ai-sdk/provider"
)

const parentCommit = "32e5ab7f1ab9e524477cc0ece04c690a89854a24"

type corpus struct {
	SchemaVersion int         `json:"schemaVersion"`
	ParentCommit  string      `json:"parentCommit"`
	Generator     generator   `json:"generator"`
	Rows          []corpusRow `json:"rows"`
}

type generator struct {
	GoVersion string `json:"goVersion"`
	Command   string `json:"command"`
}

type corpusRow struct {
	ID                  string       `json:"id"`
	Partition           string       `json:"partition"`
	MigrationProjection []migration  `json:"migrationProjection"`
	CanonicalBytes      string       `json:"canonicalBytesBase64"`
	ParentDecode        decodeResult `json:"parentDecode"`
}

type migration struct {
	Path      string `json:"path"`
	Operation string `json:"operation"`
	Value     string `json:"value,omitempty"`
}

type decodeResult struct {
	Status             string `json:"status"`
	SemanticProjection string `json:"semanticProjectionBase64,omitempty"`
	ExactError         string `json:"exactError,omitempty"`
}

type inputRow struct {
	id        string
	partition string
	migration []migration
	options   provider.CallOptions
}

func intPointer(value int) *int { return &value }
func floatPointer(value float64) *float64 { return &value }
func boolPointer(value bool) *bool { return &value }

func rawOptions(key, value string) provider.ProviderOptions {
	return provider.ProviderOptions{key: provider.RawProviderOption{Key: key, Raw: json.RawMessage(value)}}
}

		panic("output path is required")

			},
			options: provider.CallOptions{MaxOutputTokens: intPointer(9007199254740993), TopK: intPointer(-17), Seed: intPointer(0)},

				{Path: "/topK", Operation: "wrap-int", Value: "4"},

				{Path: "/includeRawChunks", Operation: "zero-bool-to-nil"},
				{Path: "/responseFormat/name", Operation: "non-empty-string-to-pointer"},

			},
		},
		{
			id: "parent-empty-collection-collapse",
			partition: "top-level/non-nil-empty-collections",
			migration: []migration{{Path: "/", Operation: "preserve-memory-but-parent-omitempty-on-encode"}},
			options: provider.CallOptions{Prompt: []provider.Message{}, Tools: []provider.Tool{}, StopSequences: []string{}, Headers: map[string]string{}, ProviderOptions: provider.ProviderOptions{}},
		},
		{
			id: "system-content-projection",
			partition: "messages/system-concatenation",

					{Type: provider.ContentPartTypeText, Text: "second", ToolName: "inactive"},
				},
			}}},
		},
		{
			id: "mixed-content-inactive-fields",
			partition: "content/mixed-inactive-fields",
			migration: []migration{
				{Path: "/prompt/0/content/0/filename", Operation: "move-file-filename-to-request-pointer", Value: "mixed.txt"},
				{Path: "/prompt/0/content/0/providerExecuted", Operation: "false-to-nil"},
				{Path: "/prompt/0/content/0/reason", Operation: "non-empty-string-to-pointer"},
			},
			options: provider.CallOptions{Prompt: []provider.Message{{Role: provider.RoleUser, Content: []provider.ContentPart{{
				Type: provider.ContentPartTypeFile, Text: "inactive", Data: &provider.DataContent{Text: "active"}, Filename: "mixed.txt", MediaType: "text/plain",
				ToolCallID: "inactive-call", ToolName: "inactive-tool", Input: json.RawMessage(`{"x":1}`), ApprovalID: "inactive-approval", Approved: &approved, Reason: "inactive-reason",
			}}}}},
		},
		{
			id: "mixed-function-provider-tool",
			partition: "tools/mixed-inactive-fields",
			migration: []migration{{Path: "/tools/0/description", Operation: "non-empty-string-to-pointer"}},
			options: provider.CallOptions{

				Filename: "data.bin", MediaType: "application/octet-stream",
			}}}}},

			}}}}},

		{

		{
			id:        "source-and-generated-filename",

				{Type: provider.ContentPartTypeSource, SourceType: provider.SourceTypeDocument, ID: "source-1", Title: "Report", MediaType: "application/pdf", Filename: "source.pdf"},

			}}}},
		},
	}

	result := corpus{
		SchemaVersion: 1,
		ParentCommit:  parentCommit,
		Generator: generator{GoVersion: runtime.Version(), Command: "test/providerwire-v4/parent-corpus/generate.sh"},
		Rows: make([]corpusRow, 0, len(rows)),
	}
	for _, input := range rows {
		body, err := providerwire.EncodeCallOptions(input.options)
		if err != nil {
			panic(fmt.Errorf("encoding %s: %w", input.id, err))
		}
		row := corpusRow{
			ID: input.id, Partition: input.partition, MigrationProjection: input.migration,
			CanonicalBytes: base64.StdEncoding.EncodeToString(body),
		}
		decoded, err := providerwire.DecodeCallOptions(body)
		if err != nil {
			row.ParentDecode = decodeResult{Status: "rejected", ExactError: err.Error()}
		} else {
			projection, err := providerwire.EncodeCallOptions(decoded)
			if err != nil {
				panic(fmt.Errorf("projecting %s: %w", input.id, err))
			}
			row.ParentDecode = decodeResult{Status: "success", SemanticProjection: base64.StdEncoding.EncodeToString(projection)}
		}
		result.Rows = append(result.Rows, row)
	}

	data, err := json.MarshalIndent(result, "", "  ")
	if err != nil {
		panic(err)
	}
	data = append(data, '\n')
	if err := os.WriteFile(os.Args[1], data, 0o644); err != nil {
		panic(err)
	}
}

func reasoningPointer(value provider.ReasoningEffort) *provider.ReasoningEffort { return &value }
